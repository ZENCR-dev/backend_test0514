## 🔍 常见错误原因分析

### 1. 必填字段缺失
```javascript
// ❌ 错误示例 - 缺少必填字段
{
  "patientInfo": { "name": "患者" },
  "medicines": [{
    "medicineId": "xxx",
    "quantity": 10
    // 缺少 dosageInstructions
  }]
  // 缺少 clinicId
}

// ✅ 正确示例 - 包含所有必填字段
{
  "clinicId": "cmc9svktq0001ugucdwn5u1ou",
  "patientInfo": { "name": "患者" },
  "medicines": [{
    "medicineId": "xxx",
    "quantity": 10,
    "dosageInstructions": "用药说明"
  }]
}
```

### 2. 字段名称错误
```javascript
// ❌ 错误的字段名
{
  "clinic_id": "xxx",              // 应该是 clinicId
  "patient": {},                   // 应该是 patientInfo
  "medicineList": [],              // 应该是 medicines
  "instructions": "xxx"            // 应该在 medicines[].dosageInstructions 中
}
```

### 3. 数据类型错误
```javascript
// ❌ 错误的数据类型
{
  "medicines": [{
    "quantity": "10",              // 应该是数字，不是字符串
    "medicineId": 123              // 应该是字符串，不是数字
  }]
}
```

### 2. 添加前端验证

```javascript
// 验证函数
function validatePrescriptionData(data) {
  const errors = [];
  
  if (!data.clinicId) {
    errors.push('诊所ID不能为空');
  }
  
  if (!data.patientInfo?.name) {
    errors.push('患者姓名不能为空');
  }
  
  if (!data.medicines || data.medicines.length === 0) {
    errors.push('至少需要添加一种药品');
  }
  
  data.medicines?.forEach((medicine, index) => {
    if (!medicine.medicineId) {
      errors.push(`第${index + 1}个药品缺少药品ID`);
    }
    if (!medicine.quantity || medicine.quantity <= 0) {
      errors.push(`第${index + 1}个药品数量必须大于0`);
    }
    if (!medicine.dosageInstructions) {
      errors.push(`第${index + 1}个药品缺少用药说明`);
    }
  });
  
  return errors;
}
```

### 3. 完整的前端调用示例

```javascript
// services/prescriptionService.ts
export const prescriptionService = {
  async create(data) {
    // 验证数据
    const errors = validatePrescriptionData(data);
    if (errors.length > 0) {
      throw new Error(errors.join(', '));
    }
    
    try {
      const response = await apiClient.post('/prescriptions', {
        clinicId: data.clinicId,
        patientInfo: {
          name: data.patientInfo.name,
          age: data.patientInfo.age,
          gender: data.patientInfo.gender,
          phone: data.patientInfo.phone,
          symptoms: data.patientInfo.symptoms,
          diagnosis: data.patientInfo.diagnosis
        },
        medicines: data.medicines.map(m => ({
          medicineId: m.medicineId,
          quantity: Number(m.quantity),  // 确保是数字
          dosageInstructions: m.dosageInstructions,
          notes: m.notes
        })),
        notes: data.notes
      });
      
      return response.data;
    } catch (error) {
      console.error('创建处方失败:', error.response?.data);
      throw error;
    }
  }
};
```
